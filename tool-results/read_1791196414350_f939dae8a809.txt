   100→/** أفضل سعر متاح لعملة عند تاريخ معين (ليومه، أو آخر سعر ≤ التاريخ، أو الأحدث مطلقاً) */
   101→export async function rateAt(
   102→  tx: Tx,
   103→  currencyId: number,
   104→  date: string
   105→): Promise<number> {
   106→  const currency = await tx.currency.findUnique({ where: { id: currencyId } })
   107→  if (!currency) throw new DomainError("العملة غير موجودة")
   108→  if (currency.isBase) return 1
   109→  const rows = await tx.exchangeRate.findMany({
   110→    where: { currencyId },
   111→    orderBy: { rateDate: "desc" },
   112→    select: { rateDate: true, rate: true },
   113→  })
   114→  if (rows.length === 0) return 1
   115→  const exact = rows.find((r) => r.rateDate === date)
   116→  if (exact) return exact.rate
   117→  const before = rows.find((r) => r.rateDate < date)
   118→  return before?.rate ?? rows[0].rate
   119→}

   123→export interface CashboxBalanceRow {

   137→/**
   138→ * رصيد صندوق واحد بعملته — Σ(حركات بعملة الصندوق أو محوّلة إليها).
   139→ * الحركات بعملة غير عملة الصندوق تُحوَّل: المبلغ×سعره → الأساس → ÷ سعر عملة الصندوق.
   140→ * (الكتابة تمنع أصلاً عدم التطابق إلا في تحويل/بنكي بين صندوقين بعملتين مختلفتين.)
   141→ */
   142→export async function computeCashboxBalance(tx: Tx, cashboxId: number): Promise<number> {
   143→  const box = await tx.cashbox.findUnique({
   144→    where: { id: cashboxId },
   145→    include: { currency: true },
   146→  })
   147→  if (!box) throw new DomainError("الصندوق غير موجود")
   148→  const rows = await tx.cashTx.findMany({
   149→    where: { OR: [{ cashboxId }, { toCashboxId: cashboxId }] },
   150→    select: {
   151→      txType: true,
   152→      cashboxId: true,
   153→      toCashboxId: true,
   154→      currencyId: true,
   155→      amount: true,
   156→      exchangeRate: true,
   157→      txDate: true,
   158→    },
   159→  })
   160→  let balance = 0
   161→  for (const r of rows) {
   162→    const sign =
   163→      r.cashboxId === cashboxId
   164→        ? (SIGN_CASHBOX[r.txType as CashTxType] ?? 0)
   165→        : r.toCashboxId === cashboxId
   166→          ? (SIGN_TO[r.txType as CashTxType] ?? 0)
   167→          : 0
   168→    if (sign === 0) continue
   169→    let amount = r.amount
   170→    if (r.currencyId !== box.currencyId) {
   171→      // حولة عملة الحركة → الأساس → عملة الصندوق
   172→      const boxRate = await rateAt(tx, box.currencyId, r.txDate)
   173→      if (boxRate <= 0) continue
   174→      amount = (r.amount * (r.exchangeRate || 1)) / boxRate
   175→    }
   176→    balance += sign * amount
   177→  }
   178→  return round2(balance)
   179→}

   181→/** كل الصناديق بأرصدتها الحية + الإجمالي بالأساس */
   182→export async function listCashboxesWithBalances(db: PrismaClient) {
   183→  return db.$transaction(async (tx) => {
   184→    const boxes = await tx.cashbox.findMany({
   185→      where: { isArchived: false },
   186→      orderBy: [{ isDefault: "desc" }, { id: "asc" }],
   187→      include: { currency: true },
   188→    })
   189→    const out: CashboxBalanceRow[] = []
   190→    for (const b of boxes) {
   191→      const [balance, last, count] = await Promise.all([
   192→        computeCashboxBalance(tx, b.id),
   193→        tx.cashTx.findFirst({
   194→          where: { OR: [{ cashboxId: b.id }, { toCashboxId: b.id }] },
   195→          orderBy: [{ txDate: "desc" }, { id: "desc" }],
   196→          select: { txDate: true },
   197→        }),
   198→        tx.cashTx.count({ where: { OR: [{ cashboxId: b.id }, { toCashboxId: b.id }] } }),
   199→      ])
   200→      out.push({
   201→        id: b.id,
   202→        name: b.name,
   203→        currencyId: b.currencyId,
   204→        currencyCode: b.currency.code,
   205→        isDefault: b.isDefault,
   206→        balance,
   207→        balanceBase: round2(toBase(balance, b.currency.isBase ? 1 : await rateAt(tx, b.currencyId, todayStr()))),
   208→        lastTxDate: last?.txDate ?? null,
   209→        txCount: count,
   210→      })
   211→    }
   212→    const totalBase = round2(out.reduce((s, b) => s + b.balanceBase, 0))
   213→    return { cashboxes: out, totalBase }
   214→  })
   215→}

   310→/**
   311→ * حفظ حركة نقدية (ذرّي) — FR-04-02/03.
   312→ * قواعد الربط: مصروف ← فئة إلزامياً، سحبية ← موظف إلزامياً،
   313→ * قبض/صرف ← طرف (عميل/مورد) مستحسن. العملة يجب أن تطابق عملة الصندوق
   314→ * (عدا التحويل/البنكي حيث تعملة الحركة = عملة المصدر ويُحوَّل للوجهة).
   315→ */
   316→export async function saveCashTx(
   317→  db: PrismaClient,
   318→  payload: SaveCashTxPayload
   319→): Promise<{ tx: CashTxDto; cashboxBalance: number; toCashboxBalance: number | null }> {
   320→  const txDate = payload.txDate || todayStr()
   321→  return db.$transaction(async (tx) => {
   322→    const amount = Number(payload.amount)
   323→    if (!(amount > 0)) throw new DomainError("المبلغ يجب أن يكون أكبر من صفر")
   324→    const txType = payload.txType
   325→    if (!CASH_TX_TYPES.includes(txType)) throw new DomainError("نوع الحركة غير معروف")
   326→
   327→    const box = await tx.cashbox.findFirst({ where: { id: payload.cashboxId, isArchived: false } })
   328→    if (!box) throw new DomainError("الصندوق غير موجود أو مؤرشف")
   329→
   330→    // الطرف المقابل (تحويل/بنكي)
   331→    let toBox: { id: number; name: string; currencyId: number } | null = null
   332→    if (payload.toCashboxId) {
   333→      const t = await tx.cashbox.findFirst({
   334→        where: { id: payload.toCashboxId, isArchived: false },
   335→      })
   336→      if (!t) throw new DomainError("الصندوق الوجهة غير موجود أو مؤرشف")
   337→      if (t.id === box.id) throw new DomainError("لا يمكن التحويل إلى نفس الصندوق")
   338→      toBox = { id: t.id, name: t.name, currencyId: t.currencyId }
   339→    }
   340→
   341→    // ─── قواعد الربط FR-04-03 ───
   342→    let categoryId: number | null = null
   343→    let categoryName: string | null = null
   344→    if (txType === "expense") {
   345→      if (!payload.expenseCategoryId) throw new DomainError("اختر فئة المصروف")
   346→      const cat = await tx.expenseCategory.findFirst({
   347→        where: { id: payload.expenseCategoryId, isArchived: false },
   348→      })
   349→      if (!cat) throw new DomainError("فئة المصروف غير موجودة")
   350→      categoryId = cat.id
   351→      categoryName = cat.name
   352→    }
   353→    let employeeId: number | null = null
   354→    let employeeName: string | null = null
   355→    if (txType === "employee_advance" || txType === "salary_batch") {
   356→      if (!payload.employeeId) throw new DomainError("اختر الموظف")
   357→      const emp = await tx.employee.findFirst({
   358→        where: { id: payload.employeeId, isArchived: false },
   359→      })
   360→      if (!emp) throw new DomainError("الموظف غير موجود أو مؤرشف")
   361→      employeeId = emp.id
   362→      employeeName = emp.name
   363→    }
   364→    let customerId: number | null = null
   365→    let customerName: string | null = null
   366→    if (payload.customerId) {
   367→      const c = await tx.customer.findFirst({ where: { id: payload.customerId, isArchived: false } })
   368→      if (!c) throw new DomainError("العميل غير موجود أو مؤرشف")
   369→      customerId = c.id
   370→      customerName = c.name
   371→    }
   372→    let supplierId: number | null = null
   373→    let supplierName: string | null = null
   374→    if (payload.supplierId) {
   375→      const s = await tx.supplier.findFirst({ where: { id: payload.supplierId, isArchived: false } })
   376→      if (!s) throw new DomainError("المورد غير موجود أو مؤرشف")
   377→      supplierId = s.id
   378→      supplierName = s.name
   379→    }
   380→
   381→    // ─── العملة والسعر ───
   382→    const base = await tx.currency.findFirst({ where: { isBase: true } })
   383→    if (!base) throw new DomainError("لا توجد عملة أساسية معرّفة")
   384→    const currencyId = payload.currencyId ?? box.currencyId
   385→    const currency = await tx.currency.findUnique({ where: { id: currencyId } })
   386→    if (!currency || !currency.isActive) throw new DomainError("العملة غير متاحة")
   387→    // إلزام تطابق عملة الحركة مع عملة الصندوق (المصدر في التحويل/البنكي)
   388→    if (currencyId !== box.currencyId) {
   389→      throw new DomainError("عملة الحركة يجب أن تطابق عملة الصندوق")
   390→    }
   391→    let exchangeRate = 1
   392→    if (!currency.isBase) {
   393→      const given = Number(payload.exchangeRate ?? 0)
   394→      if (given > 0) exchangeRate = given
   395→      else {
   396→        const row =
   397→          (await tx.exchangeRate.findUnique({
   398→            where: { currencyId_rateDate: { currencyId, rateDate: txDate } },
   399→          })) ??
   400→          (await tx.exchangeRate.findFirst({
   401→            where: { currencyId },
   402→            orderBy: { rateDate: "desc" },
   403→          }))
   404→        if (!row || row.rate <= 0)
   405→          throw new DomainError("لا يوجد سعر صرف لهذه العملة — أدخل السعر يدوياً")
   406→        exchangeRate = row.rate
   407→      }
   408→    }
   409→
   410→    // ─── منع الرصيد السالب ───
   411→    await ensureBoxNotNegative(tx, box.id, amount, txType)
   412→    if (toBox && txType === "bank_withdraw") {
   413→      // السحب البنكي: to_cashbox = حساب البنك المصدر — يُخصم منه المبلغ
   414→      const toBoxRow = await tx.cashbox.findUnique({ where: { id: toBox.id } })
   415→      if (toBoxRow) {
   416→        const toBalance = await computeCashboxBalance(tx, toBox.id)
   417→        const boxRate =
   418→          toBoxRow.currencyId === currencyId ? 1 : await rateAt(tx, toBoxRow.currencyId, txDate)
   419→        const converted = round2((amount * exchangeRate) / (boxRate > 0 ? boxRate : 1))
   420→        if (toBalance - converted < -0.01)
   421→          throw new DomainError(
   422→            `رصيد «${toBox.name}» غير كافٍ — المتاح: ${toBalance.toLocaleString("en-US")}`
   423→          )
   424→      }
   425→    }
   426→
   427→    // ─── وصف افتراضي ───
   428→    const label = TX_TYPE_LABELS[txType]
   429→    const partyLabel =
   430→      customerName ?? supplierName ?? employeeName ?? categoryName ?? null
   431→    const description =
   432→      payload.description?.trim() ||
   433→      `${label}${partyLabel ? ` — ${partyLabel}` : ""}${
   434→        txType === "box_transfer" && toBox ? ` (${box.name} ← ${toBox.name})` : ""
   435→      }`
   436→
   437→    const row = await tx.cashTx.create({
   438→      data: {
   439→        txType,
   440→        cashboxId: box.id,
   441→        toCashboxId: toBox?.id ?? null,
   442→        currencyId,
   443→        amount: round4(amount),
   444→        exchangeRate,
   445→        txDate,
   446→        refType: payload.refType ?? null,
   447→        refId: payload.refId ?? null,
   448→        expenseCategoryId: categoryId,
   449→        employeeId,
   450→        customerId,
   451→        supplierId,
   452→        description,
   453→      },
   454→      include: txInclude,
   455→    })
   456→
   457→    const cashboxBalance = await computeCashboxBalance(tx, box.id)
   458→    const toCashboxBalance = toBox ? await computeCashboxBalance(tx, toBox.id) : null
   459→    return { tx: toCashTxDto(row), cashboxBalance, toCashboxBalance }
   460→  })
   461→}

   463→/** فحص عدم سالبية رصيد الصندوق قبل أي حركة خارجة */
   464→async function ensureBoxNotNegative(
   465→  tx: Tx,
   466→  cashboxId: number,
   467→  amount: number,
   468→  txType: CashTxType
   469→): Promise<void> {
   470→  if (!isOutflowOnCashbox(txType)) return
   471→  const box = await tx.cashbox.findUnique({ where: { id: cashboxId }, include: { currency: true } })
   472→  if (!box) return
   473→  const balance = await computeCashboxBalance(tx, cashboxId)
   474→  if (balance - amount < -0.01) {
   475→    throw new DomainError(
   476→      `رصيد «${box.name}» غير كافٍ — المتاح: ${balance.toLocaleString("en-US")} ${box.currency.code}`
   477→    )
   478→  }
   479→}

   481→// ─── تعديل/حذف حركة يدوية (FR-04-08 — Task 9-a) ───
   482→
   483→/** رسالة حماية الحركات المرتبطة بمستندات */
   484→const PROTECTED_TX_MSG =
   485→  "حركة مرتبطة بمستند (فاتورة/سند/قسط/رواتب/عمولة) — تُدار من مصدرها"
   486→
   487→/**
   488→ * مساهمة حركة على رصيد صندوق معيّن بعملة ذلك الصندوق — نفس منطق
   489→ * computeCashboxBalance حرفياً (الإشارة + تحويل العملة عبر rateAt بتاريخ الحركة).
   490→ */
   491→async function txContributionOn(
   492→  tx: Tx,
   493→  row: { txType: string; currencyId: number; amount: number; exchangeRate: number },
   494→  side: "cashbox" | "to",
   495→  boxCurrencyId: number,
   496→  date: string
   497→): Promise<number> {
   498→  const sign =
   499→    side === "cashbox"
   500→      ? (SIGN_CASHBOX[row.txType as CashTxType] ?? 0)
   501→      : (SIGN_TO[row.txType as CashTxType] ?? 0)
   502→  if (sign === 0) return 0
   503→  let amount = row.amount
   504→  if (row.currencyId !== boxCurrencyId) {
   505→    const boxRate = await rateAt(tx, boxCurrencyId, date)
   506→    if (boxRate <= 0) return 0
   507→    amount = (row.amount * (row.exchangeRate || 1)) / boxRate
   508→  }
   509→  return sign * amount
   510→}

   524→/**
   525→ * تعديل حركة نقدية يدوية (FR-04-08) — الحقول القابلة للتعديل حصراً:
   526→ * amount / txDate / description (النوع والصندوق والطرف ثابتون).
   527→ * الحركات المرتبطة بمستند (refType ≠ null) محمية وتُدار من مصدرها.
   528→ * حارس الرصيد: محاكاة أثر التعديل على الصندوق (والوجهة إن وجدت) قبل الكتابة.
   529→ */
   530→export async function updateCashTx(
   531→  db: PrismaClient,
   532→  id: number,
   533→  payload: UpdateCashTxPayload
   534→): Promise<{ tx: CashTxDto; cashboxBalance: number; toCashboxBalance: number | null }> {
   535→  return db.$transaction(async (tx) => {
   536→    const row = await tx.cashTx.findUnique({
   537→      where: { id },
   538→      include: { cashbox: true, toCashbox: true, currency: true },
   539→    })
   540→    if (!row) throw new DomainError("الحركة غير موجودة", 404)
   541→    if (row.refType != null) throw new DomainError(PROTECTED_TX_MSG)
   542→
   543→    // ─── القيم الجديدة ───
   544→    const newAmount = payload.amount !== undefined ? Number(payload.amount) : row.amount
   545→    if (!(newAmount > 0)) throw new DomainError("المبلغ يجب أن يكون أكبر من صفر")
   546→    const newDate = payload.txDate !== undefined ? payload.txDate : row.txDate
   547→    if (typeof newDate !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(newDate)) {
   548→      throw new DomainError("تاريخ الحركة غير صالح (YYYY-MM-DD)")
   549→    }
   550→    const newDescription =
   551→      payload.description === undefined
   552→        ? row.description
   553→        : payload.description?.trim()
   554→          ? payload.description.trim()
   555→          : null
   556→
   557→    // ─── حارس الرصيد: محاكاة الأثر قبل الكتابة ───
   558→    const box = await tx.cashbox.findUnique({
   559→      where: { id: row.cashboxId },
   560→      include: { currency: true },
   561→    })
   562→    if (!box) throw new DomainError("الصندوق غير موجود", 404)
   563→
   564→    const oldRow = {
   565→      txType: row.txType,
   566→      currencyId: row.currencyId,
   567→      amount: row.amount,
   568→      exchangeRate: row.exchangeRate,
   569→    }
   570→    const newRow = { ...oldRow, amount: newAmount }
   571→
   572→    const balance = await computeCashboxBalance(tx, box.id)
   573→    const oldOnBox = await txContributionOn(tx, oldRow, "cashbox", box.currencyId, row.txDate)
   574→    const newOnBox = await txContributionOn(tx, newRow, "cashbox", box.currencyId, newDate)
   575→    const resultingBox = round2(balance - oldOnBox + newOnBox)
   576→    if (resultingBox < -0.01) {
   577→      throw new DomainError(
   578→        `رصيد الصندوق «${box.name}» لا يسمح — المتاح بعد التعديل: ${resultingBox.toLocaleString("en-US")} ${box.currency.code}`
   579→      )
   580→    }
   581→
   582→    // الوجهة (تحويل/بنكي بين صندوقين — قد تختلف عملتها فتُحاكى بالتحويل كما بالأرصدة)
   583→    let toBoxId: number | null = null
   584→    if (row.toCashboxId) {
   585→      const toBox = await tx.cashbox.findUnique({
   586→        where: { id: row.toCashboxId },
   587→        include: { currency: true },
   588→      })
   589→      if (toBox) {
   590→        toBoxId = toBox.id
   591→        const toBalance = await computeCashboxBalance(tx, toBox.id)
   592→        const oldOnTo = await txContributionOn(tx, oldRow, "to", toBox.currencyId, row.txDate)
   593→        const newOnTo = await txContributionOn(tx, newRow, "to", toBox.currencyId, newDate)
   594→        const resultingTo = round2(toBalance - oldOnTo + newOnTo)
   595→        if (resultingTo < -0.01) {
   596→          throw new DomainError(
   597→            `رصيد الصندوق «${toBox.name}» لا يسمح — المتاح بعد التعديل: ${resultingTo.toLocaleString("en-US")} ${toBox.currency.code}`
   598→          )
   599→        }
   600→      }
   601→    }
   602→
   603→    const updated = await tx.cashTx.update({
   604→      where: { id },
   605→      data: { amount: round4(newAmount), txDate: newDate, description: newDescription },
   606→      include: txInclude,
   607→    })
   608→    const cashboxBalance = await computeCashboxBalance(tx, box.id)
   609→    const toCashboxBalance = toBoxId ? await computeCashboxBalance(tx, toBoxId) : null
   610→    return { tx: toCashTxDto(updated), cashboxBalance, toCashboxBalance }
   611→  })
   612→}

   675→export async function listCashTx(
   676→  db: AnyDb,
   677→  filters: {
   678→    cashboxId?: number
   679→    type?: string
   680→    from?: string
   681→    to?: string
   682→    q?: string
   683→    page?: number
   684→    limit?: number
   685→  }
   686→): Promise<{ txs: CashTxDto[]; total: number; page: number; pages: number; totals: { in: number; out: number } }> {
   687→  const page = Math.max(1, filters.page ?? 1)
   688→  const limit = Math.min(100, filters.limit ?? 25)
   689→  const where: Prisma.CashTxWhereInput = {
   690→    ...(filters.cashboxId
   691→      ? { OR: [{ cashboxId: filters.cashboxId }, { toCashboxId: filters.cashboxId }] }
   692→      : {}),
   693→    ...(filters.type ? { txType: filters.type } : {}),
   694→    ...(filters.from || filters.to
   695→      ? { txDate: { ...(filters.from ? { gte: filters.from } : {}), ...(filters.to ? { lte: filters.to } : {}) } }
   696→      : {}),
   697→    ...(filters.q ? { description: { contains: filters.q } } : {}),
   698→  }
   699→  const [rows, total, inAgg, outAgg] = await Promise.all([
   700→    db.cashTx.findMany({
   701→      where,
   702→      orderBy: [{ txDate: "desc" }, { id: "desc" }],
   703→      skip: (page - 1) * limit,
   704→      take: limit,
   705→      include: txInclude,
   706→    }),
   707→    db.cashTx.count({ where }),
   708→    db.cashTx.aggregate({
   709→      _sum: { amount: true },
   710→      where: { ...where, txType: { in: ["receipt", "bank_withdraw", "opening"] } },
   711→    }),
   712→    db.cashTx.aggregate({
   713→      _sum: { amount: true },
   714→      where: {
   715→        ...where,
   716→        txType: { in: ["payment", "expense", "employee_advance", "commission_payout", "bank_deposit", "salary_batch"] },
   717→      },
   718→    }),
   719→  ])
   720→  return {
   721→    txs: rows.map(toCashTxDto),
   722→    total,
   723→    page,
   724→    pages: Math.max(1, Math.ceil(total / limit)),
   725→    totals: { in: round2(inAgg._sum.amount ?? 0), out: round2(outAgg._sum.amount ?? 0) },
   726→  }
   727→}

   776→/** حركات صندوق منذ تاريخ (تصنيفها لتقرير الوردية) */
   777→async function shiftBreakdown(tx: Tx, cashboxId: number, from: string): Promise<ShiftBreakdown> {
   778→  const rows = await tx.cashTx.findMany({
   779→    where: {
   780→      txDate: { gte: from },
   781→      OR: [{ cashboxId }, { toCashboxId: cashboxId }],
   782→    },
   783→    select: { txType: true, refType: true, cashboxId: true, toCashboxId: true, amount: true },
   784→  })
   785→  const b: ShiftBreakdown = {
   786→    cashSales: 0,
   787→    collections: 0,
   788→    otherIn: 0,
   789→    expenses: 0,
   790→    payments: 0,
   791→    advances: 0,
   792→    commissions: 0,
   793→    salaries: 0,
   794→    bankDeposits: 0,
   795→    transfersOut: 0,
   796→    transfersIn: 0,
   797→    net: 0,
   798→  }
   799→  for (const r of rows) {
   800→    // حركات تسوية الفرق (إقفال وردية) ليست نشاطاً تجارياً — تُستثنى من التصنيف
   801→    if (r.refType === "shift_reconcile") continue
   802→    const onSource = r.cashboxId === cashboxId
   803→    switch (r.txType) {
   804→      case "receipt":
   805→        if (onSource) {
   806→          if (r.refType === "invoice") b.cashSales += r.amount
   807→          else b.collections += r.amount
   808→        }
   809→        break
   810→      case "bank_withdraw":
   811→        if (onSource) b.otherIn += r.amount
   812→        break
   813→      case "opening":
   814→        if (onSource) b.otherIn += r.amount
   815→        break
   816→      case "expense":
   817→        if (onSource) b.expenses += r.amount
   818→        break
   819→      case "payment":
   820→        if (onSource) b.payments += r.amount
   821→        break
   822→      case "employee_advance":
   823→        if (onSource) b.advances += r.amount
   824→        break
   825→      case "commission_payout":
   826→        if (onSource) b.commissions += r.amount
   827→        break
   828→      case "salary_batch":
   829→        if (onSource) b.salaries += r.amount
   830→        break
   831→      case "bank_deposit":
   832→        if (onSource) b.bankDeposits += r.amount
   833→        break
   834→      case "box_transfer":
   835→        if (onSource) b.transfersOut += r.amount
   836→        else if (r.toCashboxId === cashboxId) b.transfersIn += r.amount
   837→        break
   838→    }
   839→  }
   840→  b.net =
   841→    b.cashSales +
   842→    b.collections +
   843→    b.otherIn +
   844→    b.transfersIn -
   845→    (b.expenses +
   846→      b.payments +
   847→      b.advances +
   848→      b.commissions +
   849→      b.salaries +
   850→      b.bankDeposits +
   851→      b.transfersOut)
   852→  for (const k of Object.keys(b) as Array<keyof ShiftBreakdown>) {
   853→    b[k] = round2(b[k])
   854→  }
   855→  return b
   856→}

   858→/** حالة وردية صندوق: المفتوحة + الرصيد المتوقع + التصنيف + السجل */
   859→export async function getShiftState(db: PrismaClient, cashboxId: number): Promise<ShiftStateDto> {
   860→  return db.$transaction(async (tx) => {
   861→    const box = await tx.cashbox.findUnique({ where: { id: cashboxId }, include: { currency: true } })
   862→    if (!box) throw new DomainError("الصندوق غير موجود")
   863→    const [open, history, expected] = await Promise.all([
   864→      tx.shift.findFirst({ where: { cashboxId, closedAt: null }, orderBy: { id: "desc" } }),
   865→      tx.shift.findMany({
   866→        where: { cashboxId, closedAt: { not: null } },
   867→        orderBy: { id: "desc" },
   868→        take: 20,
   869→      }),
   870→      computeCashboxBalance(tx, cashboxId),
   871→    ])
   872→    const from = open ? shiftStartDate(open.openedAt) : todayStr()
   873→    const breakdown = await shiftBreakdown(tx, cashboxId, from)
   874→    return {
   875→      cashboxId,
   876→      cashboxName: box.name,
   877→      currencyCode: box.currency.code,
   878→      openShift: open
   879→        ? { id: open.id, openedAt: open.openedAt, openingCount: open.openingCount ?? 0, notes: open.notes }
   880→        : null,
   881→      expected,
   882→      breakdown,
   883→      history: history.map((h) => ({
   884→        id: h.id,
   885→        openedAt: h.openedAt,
   886→        closedAt: h.closedAt ?? "",
   887→        openingCount: h.openingCount,
   888→        expected: h.expected,
   889→        counted: h.counted,
   890→        difference: h.difference,
   891→        notes: h.notes,
   892→      })),
   893→    }
   894→  })
   895→}

   948→    if (!box) throw new DomainError("الصندوق غير موجود")
   949→    const counted = Number(payload.counted)
   950→    if (!(counted >= 0)) throw new DomainError("أدخل العدّ الفعلي للنقدية")
   951→    let shift = await tx.shift.findFirst({ where: { cashboxId: box.id, closedAt: null }, orderBy: { id: "desc" } })
   952→    if (!shift) {
   953→      // لا وردية مفتوحة → ننشئ ونقفل فوراً (تقرير لحظي)
   954→      shift = await tx.shift.create({
   955→        data: {
   956→          cashboxId: box.id,
   957→          openedAt: new Date().toISOString(),
   958→          openingCount: await computeCashboxBalance(tx, box.id),
   959→        },
   960→      })
   961→    }
   962→    const expected = await computeCashboxBalance(tx, box.id)
   963→    const difference = round2(counted - expected)
   964→    const closedAt = new Date().toISOString()
   965→    await tx.shift.update({
   966→      where: { id: shift.id },
   967→      data: { closedAt, expected, counted, difference, notes: payload.notes ?? shift.notes },
   968→    })
   969→    // التصنيف يُحسب قبل حركة التسوية — يبقى انعكاساً لنشاط الوردية نفسها
   970→    const breakdown = await shiftBreakdown(tx, box.id, shiftStartDate(shift.openedAt))
   971→
   972→    // ─── تسوية الفرق آلياً كحركة نقدية ───
   973→    let reconciled = false
   974→    let reconcileTxId: number | null = null
   975→    if (payload.reconcile && Math.abs(difference) >= 0.01) {
   976→      const reconcileTx = await tx.cashTx.create({
   977→        data: {
   978→          txType: difference > 0 ? "receipt" : "payment",
   979→          cashboxId: box.id,
   980→          toCashboxId: null,
   981→          currencyId: box.currencyId,
   982→          amount: round2(Math.abs(difference)),
   983→          exchangeRate: box.currency.isBase ? 1 : await rateAt(tx, box.currencyId, todayStr()),
   984→          txDate: todayStr(),
   985→          refType: "shift_reconcile",
   986→          refId: shift.id,
   987→          description:
   988→            difference > 0
   989→              ? `تسوية زيادة عدّ — إقفال الوردية #${shift.id}`
   990→              : `تسوية عجز عدّ — إقفال الوردية #${shift.id}`,
   991→        },
   992→      })
   993→      reconciled = true
   994→      reconcileTxId = reconcileTx.id
   995→    }
   996→
   997→    return {
   998→      shiftId: shift.id,
   999→      cashboxName: box.name,
  1000→      currencyCode: box.currency.code,
  1001→      openedAt: shift.openedAt,
  1002→      closedAt,
  1003→      expected,
  1004→      counted,
  1005→      difference,
  1006→      breakdown,
  1007→      reconciled,
  1008→      reconcileTxId,
  1009→    }
  1010→  })
  1011→}

  1020→  totalBase: number

  1024→export async function listExpenseCategories(db: AnyDb): Promise<ExpenseCategoryDto[]> {
  1025→  const cats = await db.expenseCategory.findMany({ orderBy: { id: "asc" } })
  1026→  const out: ExpenseCategoryDto[] = []
  1027→  for (const c of cats) {
  1028→    const [txCount, agg, last] = await Promise.all([
  1029→      db.cashTx.count({ where: { expenseCategoryId: c.id } }),
  1030→      db.cashTx.aggregate({
  1031→        _sum: { amount: true },
  1032→        where: { expenseCategoryId: c.id, txType: "expense" },
  1033→      }),
  1034→      db.cashTx.findFirst({
  1035→        where: { expenseCategoryId: c.id },
  1036→        orderBy: { txDate: "desc" },
  1037→        select: { txDate: true },
  1038→      }),
  1039→    ])
  1040→    out.push({
  1041→      id: c.id,
  1042→      name: c.name,
  1043→      isArchived: c.isArchived,
  1044→      txCount,
  1045→      totalBase: round2(agg._sum.amount ?? 0),
  1046→      lastUsedAt: last?.txDate ?? null,
  1047→    })
  1048→  }
  1049→  return out
  1050→}
