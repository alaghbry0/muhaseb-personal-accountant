   224→  async function saveInvoice(
   225→    payMode: "cash" | "credit" | "mixed" | "held",
   226→    paidAmount?: number
   227→  ) {
   228→    if (!clientValidate(payMode)) return;
   229→
   230→    // فحص حد الائتمان قبل الحفظ الآجل/المختلط (FR-03-05 تحذير — ليس منعاً)
   231→    if (payMode === "credit" || payMode === "mixed") {
   232→      const cust = customersData?.customers.find((c) => c.id === pos.customerId);
   233→      if (cust && cust.creditLimit > 0) {
   234→        const due =
   235→          payMode === "credit"
   236→            ? totals.total
   237→            : Math.max(0, totals.total - (paidAmount ?? 0));
   238→        const dueBase = due * rate; // تحويل للعملة الأساسية (الرصيد والحد بها)
   239→        const balanceAfter = (cust.balance ?? 0) + dueBase;
   240→        if (balanceAfter > cust.creditLimit) {
   241→          setCreditWarn({
   242→            limit: cust.creditLimit,
   243→            balanceAfter,
   244→            dueBase,
   245→            proceed: () => {
   246→              setCreditWarn(null);
   247→              void doSaveInvoice(payMode, paidAmount);
   248→            },
   249→          });
   250→          return;
   251→        }
   252→      }
   253→    }
   254→
   255→    await doSaveInvoice(payMode, paidAmount);
   256→  }

   258→  async function doSaveInvoice(
   259→    payMode: "cash" | "credit" | "mixed" | "held",

   265→        docType: "sale",

   268→        salesRepId: pos.salesRepId,
   269→        cashboxId: pos.cashboxId,
   270→        warehouseId: pos.warehouseId,
   271→        currencyId: pos.currencyId,
   272→        exchangeRate: rate,
   273→        items: pos.lines.map((l) => ({
   274→          productId: l.productId,
   275→          qty: l.qty,
   276→          unitPrice: l.unitPrice,
   277→          discountPercent: l.discountPercent,
   278→        })),
   279→        invoiceDiscount: pos.invoiceDiscount,
   280→        taxRate: pos.taxRate,
   281→        payMode,
   282→        paidAmount: paidAmount ?? null,
   283→        replaceHeldId: pos.resumeHeldId,
   284→      });
   285→      beep(1568, 0.12);
   286→      setSuccess(res);
   287→      pos.clearAll();
   288→      setNumberInput(null);
   289→      setSearch("");
   290→      invalidateAfterSave();
   291→    } catch {
   292→      // postJson يعرض رسالة الخطأ العربية

   294→      setSaving(null);
   295→    }
   296→  }

   997→            <div className="flex gap-2.5">

  1009→          </div>
