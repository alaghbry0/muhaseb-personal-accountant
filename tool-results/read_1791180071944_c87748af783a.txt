     2→ * Domain — الأطراف (العملاء/الموردون) — Task 3-b ⭐
     3→ * أرصدة حية + سندات قبض/صرف + كشوف الحسابات — FR-03.
     4→ *
     5→ * ═══════════════ اتفاقية الأرصدة (متسقة مع Task 2 — موثقة في worklog) ═══════════════
     6→ * رصيد العميل (مدين موجب = مديون لنا):
     7→ *   opening + Σ(due فواتير البيع المكتملة × السعر)   ← الجزء الآجل وقت الإصدار (ثابت لا يُعدَّل لاحقاً)
     8→ *          − Σ(due مرتجعات البيع × السعر)             ← الجزء المُسقط من حساب العميل (ائتمان)
     9→ *          + Σ(أصل الخطط المستقلة)                     ← خطط «مبلغ مخصص» بلا فاتورة (FR-05-01)
    10→ *          − Σ(قبض من العميل بأي refType ≠ 'invoice')  ← سندات قبض حرة/مرتبطة + دفعات أولى + تحصيل أقساط
    11→ *          + Σ(صرف للعميل بأي refType ≠ 'invoice')     ← مستردات نقدية للعميل (سند صرف)
    12→ * • دفع البيع النقدي وقت الحفظ (refType='invoice') مستثنى — منعكس أصلاً في due الفاتورة (اتفاقية Task 2).
    13→ * • due الفاتورة لا يُعدَّل أبداً بعد الإصدار → تاريخ الكشف مستقر (كل تحصيل لاحق = سند قبض).
    14→ * رصيد المورد (موجب = مستحق له):
    15→ *   opening + Σ(due فواتير الشراء) − Σ(due مرتجعات الشراء)
    16→ *          − Σ(صرف للمورد refType ≠ 'invoice') + Σ(قبض من المورد refType ≠ 'invoice')
    17→ */

    28→export function todayStr(): string {
    29→  const d = new Date()
    30→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`
    31→}

    35→export async function resolveRate(
    36→  tx: Tx,
    37→  currencyId: number,
    38→  rate: number | null | undefined,
    39→  date: string
    40→): Promise<{ id: number; isBase: boolean; rate: number }> {
    41→  const currency = await tx.currency.findUnique({ where: { id: currencyId } })
    42→  if (!currency || !currency.isActive) throw new DomainError("العملة غير متاحة")
    43→  if (currency.isBase) return { id: currency.id, isBase: true, rate: 1 }
    44→  const given = Number(rate ?? 0)
    45→  if (given > 0) return { id: currency.id, isBase: false, rate: given }
    46→  const row =
    47→    (await tx.exchangeRate.findUnique({
    48→      where: { currencyId_rateDate: { currencyId: currency.id, rateDate: date } },
    49→    })) ??
    50→    (await tx.exchangeRate.findFirst({
    51→      where: { currencyId: currency.id },
    52→      orderBy: { rateDate: "desc" },
    53→    }))
    54→  if (!row || row.rate <= 0) {
    55→    throw new DomainError("لا يوجد سعر صرف لهذه العملة بتاريخ العملية — أدخل السعر يدوياً")
    56→  }
    57→  return { id: currency.id, isBase: false, rate: row.rate }
    58→}

    60→async function baseCurrency(tx: Tx): Promise<{ id: number; code: string }> {
    61→  const base = await tx.currency.findFirst({ where: { isBase: true } })
    62→  if (!base) throw new DomainError("لا توجد عملة أساسية معرّفة")
    63→  return { id: base.id, code: base.code }
    64→}

    68→/** due فواتير الطرف بالأساس: saleDue/purchaseDue و returnDue (المرتجعات) */
    69→async function partyInvoiceDue(
    70→  db: AnyDb,
    71→  party: PartyType,
    72→  partyId: number
    73→): Promise<{ purchaseDue: number; returnDue: number; movements: number }> {
    74→  const rows = await db.invoice.findMany({
    75→    where:
    76→      party === "customer"
    77→        ? { customerId: partyId, status: "completed", docType: { in: ["sale", "sale_return"] } }
    78→        : { supplierId: partyId, status: "completed", docType: { in: ["purchase", "purchase_return"] } },
    79→    select: { docType: true, dueAmount: true, exchangeRate: true },
    80→  })
    81→  let purchaseDue = 0 // «purchaseDue» = بيع آجل للعميل (موجب) أو مشتريات آجلة من المورد (موجب)
    82→  let returnDue = 0 // المرتجعات: due سالب إلزامياً (اتفاقية Task 3-a: refundMethod='credit' → due = −total)
    83→  for (const r of rows) {
    84→    const v = round4(toBase(r.dueAmount, r.exchangeRate))
    85→    if (r.docType === "sale_return" || r.docType === "purchase_return") returnDue = round4(returnDue + v)
    86→    else purchaseDue = round4(purchaseDue + v)
    87→  }
    88→  return { purchaseDue, returnDue, movements: rows.length }
    89→}

    91→/** أصل الخطط المستقلة (بلا فاتورة) — دين يُنشأ عند إنشاء الخطة (تلغى فقط إن لم يُدفع منها شيء) */
    92→async function standalonePrincipal(db: AnyDb, customerId: number) {
    93→  const plans = await db.installmentPlan.findMany({
    94→    where: { customerId, invoiceId: null },
    95→    select: { principal: true, status: true, totalPaid: true },
    96→  })
    97→  let total = 0
    98→  for (const p of plans) {
    99→    if (p.status === "cancelled" && p.totalPaid <= 0.005) continue
   100→    total = round4(total + p.principal)
   101→  }
   102→  return total
   103→}

   105→/** صافي السندات غير المرتبطة بدفع وقت الإصدار: refType ≠ 'invoice' — محوّل للأساس */
   106→async function partyVouchersBase(
   107→  db: AnyDb,
   108→  party: PartyType,
   109→  partyId: number
   110→): Promise<{ receiptsBase: number; paymentsBase: number }> {
   111→  const where =
   112→    party === "customer"
   113→      ? { customerId: partyId, refType: { not: "invoice" as const } }
   114→      : { supplierId: partyId, refType: { not: "invoice" as const } }
   115→  const [recRows, payRows] = await Promise.all([
   116→    db.cashTx.findMany({
   117→      where: { ...where, txType: "receipt" },
   118→      select: { amount: true, exchangeRate: true },
   119→    }),
   120→    db.cashTx.findMany({
   121→      where: { ...where, txType: "payment" },
   122→      select: { amount: true, exchangeRate: true },
   123→    }),
   124→  ])
   125→  let receiptsBase = 0
   126→  for (const r of recRows) receiptsBase = round4(receiptsBase + toBase(r.amount, r.exchangeRate))
   127→  let paymentsBase = 0
   128→  for (const p of payRows) paymentsBase = round4(paymentsBase + toBase(p.amount, p.exchangeRate))
   129→  return { receiptsBase, paymentsBase }
   130→}

   132→/**
   133→ * رصيد العميل الحالي (مدين موجب؛ سالب = دائن — دفعات مقدمة/مستردات).
   134→ * الدالة المرجعية لكل الشاشات (Task 4/5 يستوردونها من هنا).
   135→ */
   136→export async function computeCustomerBalance(db: AnyDb, customerId: number): Promise<number> {
   137→  const customer = await db.customer.findUnique({
   138→    where: { id: customerId },
   139→    select: { openingBalance: true },
   140→  })
   141→  if (!customer) return 0
   142→  const [inv, standalone, vouch] = await Promise.all([
   143→    partyInvoiceDue(db, "customer", customerId),
   144→    standalonePrincipal(db, customerId),
   145→    partyVouchersBase(db, "customer", customerId),
   146→  ])
   147→  return round2(
   148→    customer.openingBalance +
   149→      inv.purchaseDue +
   150→      inv.returnDue + // سالب — المرتجعات المُسقطة من حساب العميل (ائتمان)
   151→      standalone -
   152→      vouch.receiptsBase +
   153→      vouch.paymentsBase
   154→  )
   155→}

   157→/** رصيد المورد الحالي (موجب = مستحق له) */
   158→export async function computeSupplierBalance(db: AnyDb, supplierId: number): Promise<number> {
   159→  const supplier = await db.supplier.findUnique({
   160→    where: { id: supplierId },
   161→    select: { openingBalance: true },
   162→  })
   163→  if (!supplier) return 0
   164→  const [inv, vouch] = await Promise.all([
   165→    partyInvoiceDue(db, "supplier", supplierId),
   166→    partyVouchersBase(db, "supplier", supplierId),
   167→  ])
   168→  return round2(
   169→    supplier.openingBalance + inv.purchaseDue + inv.returnDue - vouch.paymentsBase + vouch.receiptsBase
   170→  )
   171→}

   201→  cashboxName: string

   211→export async function saveVoucher(
   212→  db: PrismaClient,
   213→  payload: SaveVoucherPayload
   214→): Promise<{ voucher: VoucherDto; partyBalance: number; overpayWarning: boolean }> {
   215→  const txDate = payload.txDate || todayStr()
   216→  return db.$transaction(async (tx) => {
   217→    if (!(Number(payload.amount) > 0)) throw new DomainError("مبلغ السند يجب أن يكون أكبر من صفر")
   218→    if (!payload.cashboxId) throw new DomainError("اختر الصندوق")
   219→
   220→    const cashbox = await tx.cashbox.findFirst({ where: { id: payload.cashboxId, isArchived: false } })
   221→    if (!cashbox) throw new DomainError("الصندوق غير موجود أو مؤرشف")
   222→
   223→    // الطرف
   224→    let partyName = ""
   225→    let partyPhone: string | null = null
   226→    if (payload.partyType === "customer") {
   227→      const c = await tx.customer.findFirst({ where: { id: payload.partyId, isArchived: false } })
   228→      if (!c) throw new DomainError("العميل غير موجود أو مؤرشف")
   229→      partyName = c.name
   230→      partyPhone = c.whatsapp || c.phone
   231→    } else {
   232→      const s = await tx.supplier.findFirst({ where: { id: payload.partyId, isArchived: false } })
   233→      if (!s) throw new DomainError("المورد غير موجود أو مؤرشف")
   234→      partyName = s.name
   235→      partyPhone = s.phone
   236→    }
   237→
   238→    // العملة والسعر
   239→    const base = await baseCurrency(tx)
   240→    const resolved = await resolveRate(tx, payload.currencyId ?? base.id, payload.exchangeRate ?? null, txDate)
   241→    const currencyCode = (
   242→      await tx.currency.findUnique({ where: { id: resolved.id }, select: { code: true } })
   243→    )?.code
   244→
   245→    // ربط فاتورة مفتوحة (اختياري)
   246→    let refInvoiceId: number | null = null
   247→    let refInvoiceNo: string | null = null
   248→    if (payload.refInvoiceId) {
   249→      const inv = await tx.invoice.findFirst({
   250→        where: {
   251→          id: payload.refInvoiceId,
   252→          status: "completed",
   253→          dueAmount: { gt: 0 },
   254→          ...(payload.partyType === "customer"
   255→            ? { docType: { in: ["sale", "sale_return"] } }
   256→            : { docType: { in: ["purchase", "purchase_return"] } }),
   257→          ...(payload.partyType === "customer"
   258→            ? { customerId: payload.partyId }
   259→            : { supplierId: payload.partyId }),
   260→        },
   261→        select: { id: true, invoiceNo: true },
   262→      })
   263→      if (!inv) throw new DomainError("الفاتورة المرتبطة غير موجودة أو مسددة بالكامل")
   264→      refInvoiceId = inv.id
   265→      refInvoiceNo = inv.invoiceNo
   266→    }
   267→
   268→    const kindLabel = payload.kind === "receipt" ? "قبض" : "صرف"
   269→    const row = await tx.cashTx.create({
   270→      data: {
   271→        txType: payload.kind,
   272→        cashboxId: cashbox.id,
   273→        currencyId: resolved.id,
   274→        amount: round4(payload.amount),
   275→        exchangeRate: resolved.rate,
   276→        txDate,
   277→        refType: "voucher",
   278→        refId: refInvoiceId,
   279→        customerId: payload.partyType === "customer" ? payload.partyId : null,
   280→        supplierId: payload.partyType === "supplier" ? payload.partyId : null,
   281→        description:
   282→          payload.description?.trim() ||
   283→          `سند ${kindLabel} — ${partyName}${refInvoiceNo ? ` (فاتورة ${refInvoiceNo})` : ""}`,
   284→      },
   285→      include: { cashbox: { select: { name: true } }, currency: { select: { code: true } } },
   286→    })
   287→
   288→    const partyBalance =
   289→      payload.partyType === "customer"
   290→        ? await computeCustomerBalance(tx, payload.partyId)
   291→        : await computeSupplierBalance(tx, payload.partyId)
   292→
   293→    // تحذير الدفعات المقدمة (مسموح لكن يُعلم): قبض من عميل أكبر من مديونيته / صرف لمورد أكبر من مستحقه
   294→    const overpayWarning =
   295→      (payload.partyType === "customer" && payload.kind === "receipt" && partyBalance < 0) ||
   296→      (payload.partyType === "supplier" && payload.kind === "payment" && partyBalance < 0)
   297→
   298→    const voucher: VoucherDto = {
   299→      id: row.id,
   300→      kind: payload.kind,
   301→      partyType: payload.partyType,
   302→      partyId: payload.partyId,
   303→      partyName,
   304→      partyPhone,
   305→      amount: row.amount,
   306→      currencyCode: row.currency.code,
   307→      exchangeRate: row.exchangeRate,
   308→      amountBase: round4(toBase(row.amount, row.exchangeRate)),
   309→      cashboxId: cashbox.id,
   310→      cashboxName: row.cashbox.name,
   311→      txDate: row.txDate,
   312→      createdAt: row.createdAt.toISOString(),
   313→      description: row.description,
   314→      refInvoiceId,
   315→      refInvoiceNo,
   316→      number: `V-${row.id}`,
   317→    }
   318→    return { voucher, partyBalance, overpayWarning }
   319→  })
   320→}

   322→/** قائمة سندات القبض/الصرف (refType='voucher') بفلاتر الطرف/النوع/التاريخ */
   323→export async function listVouchers(
   324→  db: PrismaClient,
   325→  filters: { kind?: string; partyType?: string; partyId?: number; from?: string; to?: string; limit?: number }
   326→): Promise<VoucherDto[]> {
   327→  const rows = await db.cashTx.findMany({
   328→    where: {
   329→      refType: "voucher",
   330→      ...(filters.kind === "receipt" || filters.kind === "payment" ? { txType: filters.kind } : {}),
   331→      ...(filters.partyType === "customer" && filters.partyId ? { customerId: filters.partyId } : {}),
   332→      ...(filters.partyType === "supplier" && filters.partyId ? { supplierId: filters.partyId } : {}),
   333→      ...(filters.from || filters.to
   334→        ? {
   335→            txDate: {
   336→              ...(filters.from ? { gte: filters.from } : {}),
   337→              ...(filters.to ? { lte: filters.to } : {}),
   338→            },
   339→          }
   340→        : {}),
   341→    },
   342→    orderBy: [{ txDate: "desc" }, { id: "desc" }],
   343→    take: Math.min(200, Math.max(1, filters.limit ?? 50)),
   344→    include: { cashbox: { select: { name: true } }, currency: { select: { code: true } } },
   345→  })
   346→
   347→  const customerIds = rows.filter((r) => r.customerId).map((r) => r.customerId!)
   348→  const supplierIds = rows.filter((r) => r.supplierId).map((r) => r.supplierId!)
   349→  const invoiceIds = rows.filter((r) => r.refId).map((r) => r.refId!)
   350→  const [customers, suppliers, invoices] = await Promise.all([
   351→    customerIds.length
   352→      ? db.customer.findMany({ where: { id: { in: customerIds } }, select: { id: true, name: true, whatsapp: true, phone: true } })
   353→      : Promise.resolve([] as Array<{ id: number; name: string; whatsapp: string | null; phone: string | null }>),
   354→    supplierIds.length
   355→      ? db.supplier.findMany({ where: { id: { in: supplierIds } }, select: { id: true, name: true, phone: true } })
   356→      : Promise.resolve([] as Array<{ id: number; name: string; phone: string | null }>),
   357→    invoiceIds.length
   358→      ? db.invoice.findMany({ where: { id: { in: invoiceIds } }, select: { id: true, invoiceNo: true } })
   359→      : Promise.resolve([] as Array<{ id: number; invoiceNo: string }>),
   360→  ])
   361→  const cMap = new Map(customers.map((c) => [c.id, c]))
   362→  const sMap = new Map(suppliers.map((s) => [s.id, s]))
   363→  const iMap = new Map(invoices.map((i) => [i.id, i.invoiceNo]))
   364→
   365→  return rows.map((r) => {
   366→    const c = r.customerId ? cMap.get(r.customerId) : undefined
   367→    const s = r.supplierId ? sMap.get(r.supplierId) : undefined
   368→    return {
   369→      id: r.id,
   370→      kind: r.txType as VoucherKind,
   371→      partyType: (r.customerId ? "customer" : "supplier") as PartyType,
   372→      partyId: (r.customerId ?? r.supplierId)!,
   373→      partyName: c?.name ?? s?.name ?? "—",
   374→      partyPhone: c ? c.whatsapp || c.phone : s?.phone ?? null,
   375→      amount: r.amount,
   376→      currencyCode: r.currency.code,
   377→      exchangeRate: r.exchangeRate,
   378→      amountBase: round4(toBase(r.amount, r.exchangeRate)),
   379→      cashboxId: r.cashboxId,
   380→      cashboxName: r.cashbox.name,
   381→      txDate: r.txDate,
   382→      createdAt: r.createdAt.toISOString(),
   383→      description: r.description,
   384→      refInvoiceId: r.refId,
   385→      refInvoiceNo: r.refId ? iMap.get(r.refId) ?? null : null,
   386→      number: `V-${r.id}`,
   387→    }
   388→  })
   389→}

   434→/** حركات الطرف كاملة بالأساس — أساس كشوف الحساب (مرتبة تصاعدياً) */
   435→export async function getPartyMovements(
   436→  db: AnyDb,
   437→  party: PartyType,
   438→  partyId: number
   439→): Promise<Movement[]> {
   440→  const out: Movement[] = []
   441→  let seq = 0
   442→  const push = (m: Omit<Movement, "seq">) => out.push({ seq: ++seq, ...m })
   443→
   444→  const partyRow =
   445→    party === "customer"
   446→      ? await db.customer.findUnique({ where: { id: partyId }, select: { id: true } })
   447→      : await db.supplier.findUnique({ where: { id: partyId }, select: { id: true } })
   448→  if (!partyRow) throw new DomainError(party === "customer" ? "العميل غير موجود" : "المورد غير موجود", 404)
   449→
   450→  // فواتير آجلة (due وقت الإصدار — ثابت تاريخياً) ومرتجعاتها
   451→  const invoices = await db.invoice.findMany({
   452→    where:
   453→      party === "customer"
   454→        ? { customerId: partyId, status: "completed", docType: { in: ["sale", "sale_return"] } }
   455→        : { supplierId: partyId, status: "completed", docType: { in: ["purchase", "purchase_return"] } },
   456→    orderBy: [{ issuedAt: "asc" }, { id: "asc" }],
   457→    select: { id: true, invoiceNo: true, docType: true, issuedAt: true, dueAmount: true, exchangeRate: true },
   458→  })
   459→  const isReturn = (d: string) => d === "sale_return" || d === "purchase_return"
   460→  for (const inv of invoices) {
   461→    // due موجب للبيع/الشراء الآجل، سالب للمرتجعات المُسقطة من الحساب (اتفاقية Task 3-a)
   462→    const v = round4(toBase(inv.dueAmount, inv.exchangeRate))
   463→    if (Math.abs(v) < 0.005) continue // مسددة نقداً وقت الإصدار — لا أثر على الحساب
   464→    const ret = isReturn(inv.docType)
   465→    push({
   466→      date: inv.issuedAt,
   467→      docType: ret
   468→        ? party === "customer"
   469→          ? "sale_return"
   470→          : "purchase_return"
   471→        : party === "customer"
   472→          ? "sale_invoice"
   473→          : "purchase_invoice",
   474→      docLabel: ret
   475→        ? party === "customer"
   476→          ? "مرتجع بيع"
   477→          : "مرتجع مشتريات"
   478→        : party === "customer"
   479→          ? "فاتورة بيع (آجل)"
   480→          : "فاتورة مشتريات (آجل)",
   481→      docNo: inv.invoiceNo,
   482→      refId: inv.id,
   483→      debit: ret ? 0 : v,
   484→      credit: ret ? Math.abs(v) : 0,
   485→    })
   486→  }
   487→
   488→  if (party === "customer") {
   489→    // خطط مستقلة — دين مخصص بلا فاتورة: الأصل كاملاً عند الإنشاء (الدفعة الأولى والتحصيلات سندات قبض)
   490→    const plans = await db.installmentPlan.findMany({
   491→      where: { customerId: partyId, invoiceId: null },
   492→      select: { id: true, principal: true, status: true, totalPaid: true, currencyId: true, createdAt: true },
   493→    })
   494→    for (const plan of plans) {
   495→      if (plan.status === "cancelled" && plan.totalPaid <= 0.005) continue
   496→      if (Math.abs(plan.principal) < 0.005) continue
   497→      push({
   498→        date: plan.createdAt.toISOString().slice(0, 10),
   499→        docType: "plan",
   500→        docLabel: "خطة تقسيط (مبلغ مخصص)",
   501→        docNo: `PLAN-${plan.id}`,
   502→        refId: plan.id,
   503→        debit: round4(plan.principal),
   504→        credit: 0,
   505→      })
   506→    }
   507→  }
   508→
   509→  // سندات القبض من الطرف (refType ≠ 'invoice') — تشمل دفعات أولى وتحصيلات أقساط
   510→  const receipts = await db.cashTx.findMany({
   511→    where:
   512→      party === "customer"
   513→        ? { customerId: partyId, txType: "receipt", refType: { not: "invoice" } }
   514→        : { supplierId: partyId, txType: "receipt", refType: { not: "invoice" } },
   515→    orderBy: [{ txDate: "asc" }, { id: "asc" }],
   516→    select: { id: true, amount: true, exchangeRate: true, txDate: true, refType: true, refId: true },
   517→  })
   518→  for (const r of receipts) {
   519→    const v = round4(toBase(r.amount, r.exchangeRate))
   520→    if (Math.abs(v) < 0.005) continue
   521→    let label = party === "customer" ? "سند قبض" : "سند قبض من مورد"
   522→    if (r.refType === "installment") {
   523→      label = "تحصيل قسط"
   524→    }
   525→    push({
   526→      date: r.txDate,
   527→      docType: "receipt_voucher",
   528→      docLabel: label,
   529→      docNo: `V-${r.id}`,
   530→      refId: r.refId,
   531→      debit: party === "supplier" ? v : 0,
   532→      credit: party === "customer" ? v : 0,
   533→    })
   534→  }
   535→
   536→  // سندات الصرف للطرف (refType ≠ 'invoice')
   537→  const payments = await db.cashTx.findMany({
   538→    where:
   539→      party === "customer"
   540→        ? { customerId: partyId, txType: "payment", refType: { not: "invoice" } }
   541→        : { supplierId: partyId, txType: "payment", refType: { not: "invoice" } },
   542→    orderBy: [{ txDate: "asc" }, { id: "asc" }],
   543→    select: { id: true, amount: true, exchangeRate: true, txDate: true, refId: true },
   544→  })
   545→  for (const p of payments) {
   546→    const v = round4(toBase(p.amount, p.exchangeRate))
   547→    if (Math.abs(v) < 0.005) continue
   548→    push({
   549→      date: p.txDate,
   550→      docType: "payment_voucher",
   551→      docLabel: party === "customer" ? "سند صرف (مسترد للعميل)" : "سند صرف",
   552→      docNo: `V-${p.id}`,
   553→      refId: p.refId,
   554→      debit: party === "customer" ? v : 0,
   555→      credit: party === "supplier" ? v : 0,
   556→    })
   557→  }
   558→
   559→  out.sort((a, b) => (a.date === b.date ? a.seq - b.seq : a.date < b.date ? -1 : 1))
   560→  return out
   561→}

   563→/**
   564→ * كشف حساب: رصيد ما قبل الفترة + صفوف الفترة (مدين/دائن/رصيد متحرك) + إجماليات.
   565→ * from/to اختيارية. بدون from: يُدرج سطر «رصيد افتتاحي» بالرصيد الرأسي للطرف.
   566→ */
   567→export async function getStatement(
   568→  db: AnyDb,
   569→  opts: { partyType: PartyType; partyId: number; from?: string; to?: string }
   570→): Promise<StatementResult> {
   571→  const { partyType, partyId } = opts
   572→  const from = opts.from || ""
   573→  const to = opts.to || ""
   574→
   575→  const partyRow =
   576→    partyType === "customer"
   577→      ? await db.customer.findUnique({ where: { id: partyId }, select: { name: true, openingBalance: true } })
   578→      : await db.supplier.findUnique({ where: { id: partyId }, select: { name: true, openingBalance: true } })
   579→  if (!partyRow) {
   580→    throw new DomainError(partyType === "customer" ? "العميل غير موجود" : "المورد غير موجود", 404)
   581→  }
   582→
   583→  const movements = await getPartyMovements(db, partyType, partyId)
   584→
   585→  // الرصيد أول الفترة: الرأسي + صافي كل الحركات قبل from
   586→  // (بدون from = كل الحركات داخل الفترة — الافتتاحي الرأسي فقط)
   587→  let openingAtFrom = partyRow.openingBalance
   588→  if (from) {
   589→    for (const m of movements) {
   590→      if (m.date >= from) break
   591→      openingAtFrom = round4(openingAtFrom + m.debit - m.credit)
   592→    }
   593→  }
   594→
   595→  const rows: StatementRow[] = []
   596→  let running = openingAtFrom
   597→
   598→  // بدون from: نعرض الرصيد الرأسي كسطر افتتاحي
   599→  if (!from) {
   600→    rows.push({
   601→      date: "",
   602→      docType: "opening",
   603→      docLabel: "رصيد افتتاحي",
   604→      docNo: null,
   605→      refId: null,
   606→      debit: partyRow.openingBalance > 0 ? partyRow.openingBalance : 0,
   607→      credit: partyRow.openingBalance < 0 ? -partyRow.openingBalance : 0,
   608→      balance: partyRow.openingBalance,
   609→    })
   610→  }
   611→
   612→  const inRange = movements.filter((m) => (!from || m.date >= from) && (!to || m.date <= to))
   613→  for (const m of inRange) {
   614→    running = round4(running + m.debit - m.credit)
   615→    rows.push({
   616→      date: m.date,
   617→      docType: m.docType,
   618→      docLabel: m.docLabel,
   619→      docNo: m.docNo,
   620→      refId: m.refId,
   621→      debit: m.debit,
   622→      credit: m.credit,
   623→      balance: running,
   624→    })
   625→  }
   626→
   627→  const totals = rows.reduce(
   628→    (acc, r) => ({ debit: round4(acc.debit + r.debit), credit: round4(acc.credit + r.credit) }),
   629→    { debit: 0, credit: 0 }
   630→  )
   631→  const base = await db.currency.findFirst({ where: { isBase: true }, select: { code: true } })
   632→
   633→  return {
   634→    partyType,
   635→    partyId,
   636→    partyName: partyRow.name,
   637→    from,
   638→    to,
   639→    openingBalance: openingAtFrom,
   640→    rows,
   641→    totals,
   642→    closingBalance: rows.length ? rows[rows.length - 1].balance! : openingAtFrom,
   643→    baseCurrencyCode: base?.code ?? "YER",
   644→    transactionsCount: inRange.length,
   645→  }
   646→}

   678→export async function getCustomerFile(db: AnyDb, customerId: number) {
   679→  const customer = await db.customer.findUnique({ where: { id: customerId } })
   680→  if (!customer) throw new DomainError("العميل غير موجود", 404)
   681→  const balance = await computeCustomerBalance(db, customerId)
   682→
   683→  const [salesAgg, lastInvoices, plans] = await Promise.all([
   684→    db.invoice.aggregate({
   685→      _sum: { totalBase: true },
   686→      _count: true,
   687→      where: { customerId, docType: "sale", status: "completed" },
   688→    }),
   689→    db.invoice.findMany({
   690→      where: { customerId, docType: "sale", status: "completed" },
   691→      orderBy: [{ issuedAt: "desc" }, { id: "desc" }],
   692→      take: 5,
   693→      select: {
   694→        id: true, invoiceNo: true, issuedAt: true, payStatus: true,
   695→        total: true, dueAmount: true,
   696→        currency: { select: { code: true } },
   697→      },
   698→    }),
   699→    db.installmentPlan.findMany({
   700→      where: { customerId, status: { in: ["active", "defaulted"] } },
   701→      include: { installments: true, invoice: { select: { invoiceNo: true } } },
   702→    }),
   703→  ])
   704→
   705→  const today = todayStr()
   706→  const overdue: Array<{
   707→    planId: number
   708→    seq: number
   709→    installmentId: number
   710→    dueDate: string
   711→    amount: number
   712→    remaining: number
   713→    invoiceNo: string | null
   714→  }> = []
   715→  for (const plan of plans) {
   716→    for (const inst of plan.installments) {
   717→      if ((inst.status === "pending" || inst.status === "partial") && inst.dueDate < today) {
   718→        overdue.push({
   719→          planId: plan.id,
   720→          seq: inst.seq,
   721→          installmentId: inst.id,
   722→          dueDate: inst.dueDate,
   723→          amount: inst.amount,
   724→          remaining: round2(inst.amount - inst.paidAmount),
   725→          invoiceNo: plan.invoice?.invoiceNo ?? null,
   726→        })
   727→      }
   728→    }
   729→  }
   730→  overdue.sort((a, b) => (a.dueDate === b.dueDate ? a.seq - b.seq : a.dueDate < b.dueDate ? -1 : 1))
   731→
   732→  return {
   733→    customer: {
   734→      id: customer.id,
   735→      name: customer.name,
   736→      phone: customer.phone,
   737→      whatsapp: customer.whatsapp,
   738→      address: customer.address,
   739→      area: customer.area,
   740→      creditLimit: customer.creditLimit,
   741→      openingBalance: customer.openingBalance,
   742→      notes: customer.notes,
   743→      isArchived: customer.isArchived,
   744→      balance,
   745→      overLimit: customer.creditLimit > 0 && balance > customer.creditLimit,
   746→    } satisfies CustomerDto,
   747→    stats: {
   748→      salesTotalBase: round4(salesAgg._sum.totalBase ?? 0),
   749→      invoicesCount: salesAgg._count,
   750→    },
   751→    lastInvoices: lastInvoices.map((inv) => ({
   752→      id: inv.id,
   753→      invoiceNo: inv.invoiceNo,
   754→      issuedAt: inv.issuedAt,
   755→      payStatus: inv.payStatus,
   756→      total: inv.total,
   757→      dueAmount: inv.dueAmount,
   758→      currencyCode: inv.currency.code,
   759→    })),
   760→    overdueInstallments: overdue,
   761→  }
   762→}
