   100→/** أفضل سعر متاح لعملة عند تاريخ معين (ليومه، أو آخر سعر ≤ التاريخ، أو الأحدث مطلقاً) */
   101→export async function rateAt(
   102→  tx: Tx,
   103→  currencyId: number,
   104→  date: string
   105→): Promise<number> {
   106→  const currency = await tx.currency.findUnique({ where: { id: currencyId } })
   107→  if (!currency) throw new DomainError("العملة غير موجودة")
   108→  if (currency.isBase) return 1
   109→  const rows = await tx.exchangeRate.findMany({
   110→    where: { currencyId },
   111→    orderBy: { rateDate: "desc" },
   112→    select: { rateDate: true, rate: true },
   113→  })
   114→  if (rows.length === 0) return 1
   115→  const exact = rows.find((r) => r.rateDate === date)
   116→  if (exact) return exact.rate
   117→  const before = rows.find((r) => r.rateDate < date)
   118→  return before?.rate ?? rows[0].rate
   119→}

   123→export interface CashboxBalanceRow {

   137→/**
   138→ * رصيد صندوق واحد بعملته — Σ(حركات بعملة الصندوق أو محوّلة إليها).
   139→ * الحركات بعملة غير عملة الصندوق تُحوَّل: المبلغ×سعره → الأساس → ÷ سعر عملة الصندوق.
   140→ * (الكتابة تمنع أصلاً عدم التطابق إلا في تحويل/بنكي بين صندوقين بعملتين مختلفتين.)
   141→ */
   142→export async function computeCashboxBalance(tx: Tx, cashboxId: number): Promise<number> {
   143→  const box = await tx.cashbox.findUnique({
   144→    where: { id: cashboxId },
   145→    include: { currency: true },
   146→  })
   147→  if (!box) throw new DomainError("الصندوق غير موجود")
   148→  const rows = await tx.cashTx.findMany({
   149→    where: { OR: [{ cashboxId }, { toCashboxId: cashboxId }] },
   150→    select: {
   151→      txType: true,
   152→      cashboxId: true,
   153→      toCashboxId: true,
   154→      currencyId: true,
   155→      amount: true,
   156→      exchangeRate: true,
   157→      txDate: true,
   158→    },
   159→  })
   160→  let balance = 0
   161→  for (const r of rows) {
   162→    const sign =
   163→      r.cashboxId === cashboxId
   164→        ? (SIGN_CASHBOX[r.txType as CashTxType] ?? 0)
   165→        : r.toCashboxId === cashboxId
   166→          ? (SIGN_TO[r.txType as CashTxType] ?? 0)
   167→          : 0
   168→    if (sign === 0) continue
   169→    let amount = r.amount
   170→    if (r.currencyId !== box.currencyId) {
   171→      // حولة عملة الحركة → الأساس → عملة الصندوق
   172→      const boxRate = await rateAt(tx, box.currencyId, r.txDate)
   173→      if (boxRate <= 0) continue
   174→      amount = (r.amount * (r.exchangeRate || 1)) / boxRate
   175→    }
   176→    balance += sign * amount
   177→  }
   178→  return round2(balance)
   179→}

   181→/** كل الصناديق بأرصدتها الحية + الإجمالي بالأساس */
   182→export async function listCashboxesWithBalances(db: PrismaClient) {
   183→  return db.$transaction(async (tx) => {
   184→    const boxes = await tx.cashbox.findMany({
   185→      where: { isArchived: false },
   186→      orderBy: [{ isDefault: "desc" }, { id: "asc" }],
   187→      include: { currency: true },
   188→    })
   189→    const out: CashboxBalanceRow[] = []
   190→    for (const b of boxes) {
   191→      const [balance, last, count] = await Promise.all([
   192→        computeCashboxBalance(tx, b.id),
   193→        tx.cashTx.findFirst({
   194→          where: { OR: [{ cashboxId: b.id }, { toCashboxId: b.id }] },
   195→          orderBy: [{ txDate: "desc" }, { id: "desc" }],
   196→          select: { txDate: true },
   197→        }),
   198→        tx.cashTx.count({ where: { OR: [{ cashboxId: b.id }, { toCashboxId: b.id }] } }),
   199→      ])
   200→      out.push({
   201→        id: b.id,
   202→        name: b.name,
   203→        currencyId: b.currencyId,
   204→        currencyCode: b.currency.code,
   205→        isDefault: b.isDefault,
   206→        balance,
   207→        balanceBase: round2(toBase(balance, b.currency.isBase ? 1 : await rateAt(tx, b.currencyId, todayStr()))),
   208→        lastTxDate: last?.txDate ?? null,
   209→        txCount: count,
   210→      })
   211→    }
   212→    const totalBase = round2(out.reduce((s, b) => s + b.balanceBase, 0))
   213→    return { cashboxes: out, totalBase }
   214→  })
   215→}

   310→/**
   311→ * حفظ حركة نقدية (ذرّي) — FR-04-02/03.
   312→ * قواعد الربط: مصروف ← فئة إلزامياً، سحبية ← موظف إلزامياً،
   313→ * قبض/صرف ← طرف (عميل/مورد) مستحسن. العملة يجب أن تطابق عملة الصندوق
   314→ * (عدا التحويل/البنكي حيث تعملة الحركة = عملة المصدر ويُحوَّل للوجهة).
   315→ */
   316→export async function saveCashTx(
   317→  db: PrismaClient,
   318→  payload: SaveCashTxPayload
   319→): Promise<{ tx: CashTxDto; cashboxBalance: number; toCashboxBalance: number | null }> {
   320→  const txDate = payload.txDate || todayStr()
   321→  return db.$transaction(async (tx) => {
   322→    const amount = Number(payload.amount)
   323→    if (!(amount > 0)) throw new DomainError("المبلغ يجب أن يكون أكبر من صفر")
   324→    const txType = payload.txType
   325→    if (!CASH_TX_TYPES.includes(txType)) throw new DomainError("نوع الحركة غير معروف")
   326→
   327→    const box = await tx.cashbox.findFirst({ where: { id: payload.cashboxId, isArchived: false } })
   328→    if (!box) throw new DomainError("الصندوق غير موجود أو مؤرشف")
   329→
   330→    // الطرف المقابل (تحويل/بنكي)
   331→    let toBox: { id: number; name: string; currencyId: number } | null = null
   332→    if (payload.toCashboxId) {
   333→      const t = await tx.cashbox.findFirst({
   334→        where: { id: payload.toCashboxId, isArchived: false },
   335→      })
   336→      if (!t) throw new DomainError("الصندوق الوجهة غير موجود أو مؤرشف")
   337→      if (t.id === box.id) throw new DomainError("لا يمكن التحويل إلى نفس الصندوق")
   338→      toBox = { id: t.id, name: t.name, currencyId: t.currencyId }
   339→    }
   340→
   341→    // ─── قواعد الربط FR-04-03 ───
   342→    let categoryId: number | null = null
   343→    let categoryName: string | null = null
   344→    if (txType === "expense") {
   345→      if (!payload.expenseCategoryId) throw new DomainError("اختر فئة المصروف")
   346→      const cat = await tx.expenseCategory.findFirst({
   347→        where: { id: payload.expenseCategoryId, isArchived: false },
   348→      })
   349→      if (!cat) throw new DomainError("فئة المصروف غير موجودة")
   350→      categoryId = cat.id
   351→      categoryName = cat.name
   352→    }
   353→    let employeeId: number | null = null
   354→    let employeeName: string | null = null
   355→    if (txType === "employee_advance" || txType === "salary_batch") {
   356→      if (!payload.employeeId) throw new DomainError("اختر الموظف")
   357→      const emp = await tx.employee.findFirst({
   358→        where: { id: payload.employeeId, isArchived: false },
   359→      })
   360→      if (!emp) throw new DomainError("الموظف غير موجود أو مؤرشف")
   361→      employeeId = emp.id
   362→      employeeName = emp.name
   363→    }
   364→    let customerId: number | null = null
   365→    let customerName: string | null = null
   366→    if (payload.customerId) {
   367→      const c = await tx.customer.findFirst({ where: { id: payload.customerId, isArchived: false } })
   368→      if (!c) throw new DomainError("العميل غير موجود أو مؤرشف")
   369→      customerId = c.id
   370→      customerName = c.name
   371→    }
   372→    let supplierId: number | null = null
   373→    let supplierName: string | null = null
   374→    if (payload.supplierId) {
   375→      const s = await tx.supplier.findFirst({ where: { id: payload.supplierId, isArchived: false } })
   376→      if (!s) throw new DomainError("المورد غير موجود أو مؤرشف")
   377→      supplierId = s.id
   378→      supplierName = s.name
   379→    }
   380→
   381→    // ─── العملة والسعر ───
   382→    const base = await tx.currency.findFirst({ where: { isBase: true } })
   383→    if (!base) throw new DomainError("لا توجد عملة أساسية معرّفة")
   384→    const currencyId = payload.currencyId ?? box.currencyId
   385→    const currency = await tx.currency.findUnique({ where: { id: currencyId } })
   386→    if (!currency || !currency.isActive) throw new DomainError("العملة غير متاحة")
   387→    // إلزام تطابق عملة الحركة مع عملة الصندوق (المصدر في التحويل/البنكي)
   388→    if (currencyId !== box.currencyId) {
   389→      throw new DomainError("عملة الحركة يجب أن تطابق عملة الصندوق")
   390→    }
   391→    let exchangeRate = 1
   392→    if (!currency.isBase) {
   393→      const given = Number(payload.exchangeRate ?? 0)
   394→      if (given > 0) exchangeRate = given
   395→      else {
   396→        const row =
   397→          (await tx.exchangeRate.findUnique({
   398→            where: { currencyId_rateDate: { currencyId, rateDate: txDate } },
   399→          })) ??
   400→          (await tx.exchangeRate.findFirst({
   401→            where: { currencyId },
   402→            orderBy: { rateDate: "desc" },
   403→          }))
   404→        if (!row || row.rate <= 0)
   405→          throw new DomainError("لا يوجد سعر صرف لهذه العملة — أدخل السعر يدوياً")
   406→        exchangeRate = row.rate
   407→      }
   408→    }
   409→
   410→    // ─── منع الرصيد السالب ───
   411→    await ensureBoxNotNegative(tx, box.id, amount, txType)
   412→    if (toBox && txType === "bank_withdraw") {
   413→      // السحب البنكي: to_cashbox = حساب البنك المصدر — يُخصم منه المبلغ
   414→      const toBoxRow = await tx.cashbox.findUnique({ where: { id: toBox.id } })
   415→      if (toBoxRow) {
   416→        const toBalance = await computeCashboxBalance(tx, toBox.id)
   417→        const boxRate =
   418→          toBoxRow.currencyId === currencyId ? 1 : await rateAt(tx, toBoxRow.currencyId, txDate)
   419→        const converted = round2((amount * exchangeRate) / (boxRate > 0 ? boxRate : 1))
   420→        if (toBalance - converted < -0.01)
   421→          throw new DomainError(
   422→            `رصيد «${toBox.name}» غير كافٍ — المتاح: ${toBalance.toLocaleString("en-US")}`
   423→          )
   424→      }
   425→    }
   426→
   427→    // ─── وصف افتراضي ───
   428→    const label = TX_TYPE_LABELS[txType]
   429→    const partyLabel =
   430→      customerName ?? supplierName ?? employeeName ?? categoryName ?? null
   431→    const description =
   432→      payload.description?.trim() ||
   433→      `${label}${partyLabel ? ` — ${partyLabel}` : ""}${
   434→        txType === "box_transfer" && toBox ? ` (${box.name} ← ${toBox.name})` : ""
   435→      }`
   436→
   437→    const row = await tx.cashTx.create({
   438→      data: {
   439→        txType,
   440→        cashboxId: box.id,
   441→        toCashboxId: toBox?.id ?? null,
   442→        currencyId,
   443→        amount: round4(amount),
   444→        exchangeRate,
   445→        txDate,
   446→        refType: payload.refType ?? null,
   447→        refId: payload.refId ?? null,
   448→        expenseCategoryId: categoryId,
   449→        employeeId,
   450→        customerId,
   451→        supplierId,
   452→        description,
   453→      },
   454→      include: txInclude,
   455→    })
   456→
   457→    const cashboxBalance = await computeCashboxBalance(tx, box.id)
   458→    const toCashboxBalance = toBox ? await computeCashboxBalance(tx, toBox.id) : null
   459→    return { tx: toCashTxDto(row), cashboxBalance, toCashboxBalance }
   460→  })
   461→}

   463→/** فحص عدم سالبية رصيد الصندوق قبل أي حركة خارجة */
   464→async function ensureBoxNotNegative(
   465→  tx: Tx,
   466→  cashboxId: number,
   467→  amount: number,
   468→  txType: CashTxType
   469→): Promise<void> {
   470→  if (!isOutflowOnCashbox(txType)) return
   471→  const box = await tx.cashbox.findUnique({ where: { id: cashboxId }, include: { currency: true } })
   472→  if (!box) return
   473→  const balance = await computeCashboxBalance(tx, cashboxId)
   474→  if (balance - amount < -0.01) {
   475→    throw new DomainError(
   476→      `رصيد «${box.name}» غير كافٍ — المتاح: ${balance.toLocaleString("en-US")} ${box.currency.code}`
   477→    )
   478→  }
   479→}

   483→export async function listCashTx(
   484→  db: AnyDb,
   485→  filters: {
   486→    cashboxId?: number
   487→    type?: string
   488→    from?: string
   489→    to?: string
   490→    q?: string
   491→    page?: number
   492→    limit?: number
   493→  }
   494→): Promise<{ txs: CashTxDto[]; total: number; page: number; pages: number; totals: { in: number; out: number } }> {
   495→  const page = Math.max(1, filters.page ?? 1)
   496→  const limit = Math.min(100, filters.limit ?? 25)
   497→  const where: Prisma.CashTxWhereInput = {
   498→    ...(filters.cashboxId
   499→      ? { OR: [{ cashboxId: filters.cashboxId }, { toCashboxId: filters.cashboxId }] }
   500→      : {}),
   501→    ...(filters.type ? { txType: filters.type } : {}),
   502→    ...(filters.from || filters.to
   503→      ? { txDate: { ...(filters.from ? { gte: filters.from } : {}), ...(filters.to ? { lte: filters.to } : {}) } }
   504→      : {}),
   505→    ...(filters.q ? { description: { contains: filters.q } } : {}),
   506→  }
   507→  const [rows, total, inAgg, outAgg] = await Promise.all([
   508→    db.cashTx.findMany({
   509→      where,
   510→      orderBy: [{ txDate: "desc" }, { id: "desc" }],
   511→      skip: (page - 1) * limit,
   512→      take: limit,
   513→      include: txInclude,
   514→    }),
   515→    db.cashTx.count({ where }),
   516→    db.cashTx.aggregate({
   517→      _sum: { amount: true },
   518→      where: { ...where, txType: { in: ["receipt", "bank_withdraw", "opening"] } },
   519→    }),
   520→    db.cashTx.aggregate({
   521→      _sum: { amount: true },
   522→      where: {
   523→        ...where,
   524→        txType: { in: ["payment", "expense", "employee_advance", "commission_payout", "bank_deposit", "salary_batch"] },
   525→      },
   526→    }),
   527→  ])
   528→  return {
   529→    txs: rows.map(toCashTxDto),
   530→    total,
   531→    page,
   532→    pages: Math.max(1, Math.ceil(total / limit)),
   533→    totals: { in: round2(inAgg._sum.amount ?? 0), out: round2(outAgg._sum.amount ?? 0) },
   534→  }
   535→}

   584→/** حركات صندوق منذ تاريخ (تصنيفها لتقرير الوردية) */
   585→async function shiftBreakdown(tx: Tx, cashboxId: number, from: string): Promise<ShiftBreakdown> {
   586→  const rows = await tx.cashTx.findMany({
   587→    where: {
   588→      txDate: { gte: from },
   589→      OR: [{ cashboxId }, { toCashboxId: cashboxId }],
   590→    },
   591→    select: { txType: true, refType: true, cashboxId: true, toCashboxId: true, amount: true },
   592→  })
   593→  const b: ShiftBreakdown = {
   594→    cashSales: 0,
   595→    collections: 0,
   596→    otherIn: 0,
   597→    expenses: 0,
   598→    payments: 0,
   599→    advances: 0,
   600→    commissions: 0,
   601→    salaries: 0,
   602→    bankDeposits: 0,
   603→    transfersOut: 0,
   604→    transfersIn: 0,
   605→    net: 0,
   606→  }
   607→  for (const r of rows) {
   608→    const onSource = r.cashboxId === cashboxId
   609→    switch (r.txType) {
   610→      case "receipt":
   611→        if (onSource) {
   612→          if (r.refType === "invoice") b.cashSales += r.amount
   613→          else b.collections += r.amount
   614→        }
   615→        break
   616→      case "bank_withdraw":
   617→        if (onSource) b.otherIn += r.amount
   618→        break
   619→      case "opening":
   620→        if (onSource) b.otherIn += r.amount
   621→        break
   622→      case "expense":
   623→        if (onSource) b.expenses += r.amount
   624→        break
   625→      case "payment":
   626→        if (onSource) b.payments += r.amount
   627→        break
   628→      case "employee_advance":
   629→        if (onSource) b.advances += r.amount
   630→        break
   631→      case "commission_payout":
   632→        if (onSource) b.commissions += r.amount
   633→        break
   634→      case "salary_batch":
   635→        if (onSource) b.salaries += r.amount
   636→        break
   637→      case "bank_deposit":
   638→        if (onSource) b.bankDeposits += r.amount
   639→        break
   640→      case "box_transfer":
   641→        if (onSource) b.transfersOut += r.amount
   642→        else if (r.toCashboxId === cashboxId) b.transfersIn += r.amount
   643→        break
   644→    }
   645→  }
   646→  b.net =
   647→    b.cashSales +
   648→    b.collections +
   649→    b.otherIn +
   650→    b.transfersIn -
   651→    (b.expenses +
   652→      b.payments +
   653→      b.advances +
   654→      b.commissions +
   655→      b.salaries +
   656→      b.bankDeposits +
   657→      b.transfersOut)
   658→  for (const k of Object.keys(b) as Array<keyof ShiftBreakdown>) {
   659→    b[k] = round2(b[k])
   660→  }
   661→  return b
   662→}

   664→/** حالة وردية صندوق: المفتوحة + الرصيد المتوقع + التصنيف + السجل */
   665→export async function getShiftState(db: PrismaClient, cashboxId: number): Promise<ShiftStateDto> {
   666→  return db.$transaction(async (tx) => {
   667→    const box = await tx.cashbox.findUnique({ where: { id: cashboxId }, include: { currency: true } })
   668→    if (!box) throw new DomainError("الصندوق غير موجود")
   669→    const [open, history, expected] = await Promise.all([
   670→      tx.shift.findFirst({ where: { cashboxId, closedAt: null }, orderBy: { id: "desc" } }),
   671→      tx.shift.findMany({
   672→        where: { cashboxId, closedAt: { not: null } },
   673→        orderBy: { id: "desc" },
   674→        take: 20,
   675→      }),
   676→      computeCashboxBalance(tx, cashboxId),
   677→    ])
   678→    const from = open ? shiftStartDate(open.openedAt) : todayStr()
   679→    const breakdown = await shiftBreakdown(tx, cashboxId, from)
   680→    return {
   681→      cashboxId,
   682→      cashboxName: box.name,
   683→      currencyCode: box.currency.code,
   684→      openShift: open
   685→        ? { id: open.id, openedAt: open.openedAt, openingCount: open.openingCount ?? 0, notes: open.notes }
   686→        : null,
   687→      expected,
   688→      breakdown,
   689→      history: history.map((h) => ({
   690→        id: h.id,
   691→        openedAt: h.openedAt,
   692→        closedAt: h.closedAt ?? "",
   693→        openingCount: h.openingCount,
   694→        expected: h.expected,
   695→        counted: h.counted,
   696→        difference: h.difference,
   697→        notes: h.notes,
   698→      })),
   699→    }
   700→  })
   701→}

   703→/** فتح وردية — openingCount افتراضياً الرصيد الحالي */
   704→export async function openShift(
   705→  db: PrismaClient,
   706→  payload: { cashboxId: number; openingCount?: number; notes?: string }
   707→) {
   708→  return db.$transaction(async (tx) => {
   709→    const box = await tx.cashbox.findUnique({ where: { id: payload.cashboxId } })
   710→    if (!box) throw new DomainError("الصندوق غير موجود")
   711→    const existing = await tx.shift.findFirst({ where: { cashboxId: box.id, closedAt: null } })
   712→    if (existing) throw new DomainError("هناك وردية مفتوحة لهذا الصندوق بالفعل")
   713→    const balance = await computeCashboxBalance(tx, box.id)
   714→    const row = await tx.shift.create({
   715→      data: {
   716→        cashboxId: box.id,
   717→        openedAt: new Date().toISOString(),
   718→        openingCount: payload.openingCount != null ? payload.openingCount : balance,
   719→        notes: payload.notes ?? null,
   720→      },
   721→    })
   722→    return { shift: row, balance }
   723→  })
   724→}

   738→/** إقفال الوردية — FR-04-04: المتوقع (الرصيد المحسوب) مقابل العدّ الفعلي */
   739→export async function closeShift(
   740→  db: PrismaClient,
   741→  payload: { cashboxId: number; counted: number; notes?: string }
   742→): Promise<CloseShiftResult> {
   743→  return db.$transaction(async (tx) => {
   744→    const box = await tx.cashbox.findUnique({ where: { id: payload.cashboxId }, include: { currency: true } })
   745→    if (!box) throw new DomainError("الصندوق غير موجود")
   746→    const counted = Number(payload.counted)
   747→    if (!(counted >= 0)) throw new DomainError("أدخل العدّ الفعلي للنقدية")
   748→    let shift = await tx.shift.findFirst({ where: { cashboxId: box.id, closedAt: null }, orderBy: { id: "desc" } })
   749→    if (!shift) {
   750→      // لا وردية مفتوحة → ننشئ ونقفل فوراً (تقرير لحظي)
   751→      shift = await tx.shift.create({
   752→        data: {
   753→          cashboxId: box.id,
   754→          openedAt: new Date().toISOString(),
   755→          openingCount: await computeCashboxBalance(tx, box.id),
   756→        },
   757→      })
   758→    }
   759→    const expected = await computeCashboxBalance(tx, box.id)
   760→    const difference = round2(counted - expected)
   761→    const closedAt = new Date().toISOString()
   762→    await tx.shift.update({
   763→      where: { id: shift.id },
   764→      data: { closedAt, expected, counted, difference, notes: payload.notes ?? shift.notes },
   765→    })
   766→    const breakdown = await shiftBreakdown(tx, box.id, shiftStartDate(shift.openedAt))
   767→    return {
   768→      shiftId: shift.id,
   769→      cashboxName: box.name,
   770→      currencyCode: box.currency.code,
   771→      openedAt: shift.openedAt,
   772→      closedAt,
   773→      expected,
   774→      counted,
   775→      difference,
   776→      breakdown,
   777→    }
   778→  })
   779→}
